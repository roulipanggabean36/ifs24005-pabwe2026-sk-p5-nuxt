export const authApi = {
  login: (payload: LoginPayload) => api.post<LoginData>('/auth/login', payload, { auth: false }),
  register: (payload: RegisterPayload) =>

  logout: () => api.post<undefined>('/auth/logout'),
}


===== src\features\auth\authStore.ts =====

import { defineStore } from 'pinia'
import { computed, ref } from 'vue'
import { authApi } from '~/features/auth/authApi'
import type { LoginPayload, RegisterPayload, User } from '~/features/auth/types'
import { getToken, removeToken, setToken } from '~/helpers/apiHelper'

export const useAuthStore = defineStore('auth', () => {
  const token = ref<string | null>(getToken())
  const user = ref<User | null>(null)
  const loading = ref(false)

  const isAuthenticated = computed(() => token.value !== null)

  // Menyelaraskan state dengan localStorage (token dihapus otomatis oleh apiHelper saat 401)
  const syncToken = (): void => {
    token.value = getToken()
    if (token.value === null) user.value = null
  }

  const clearSession = (): void => {
    removeToken()
    token.value = null
    user.value = null
  }

  const login = async (payload: LoginPayload): Promise<void> => {
    loading.value = true
    try {
      const response = await authApi.login(payload)
      setToken(response.data.token)
      token.value = response.data.token
      user.value = response.data.user
    } finally {
      loading.value = false
    }
  }

  const register = async (payload: RegisterPayload): Promise<void> => {
    loading.value = true
    try {
      await authApi.register(payload)
    } finally {
      loading.value = false
    }
  }

  const logout = async (): Promise<void> => {
    loading.value = true
    try {
      await authApi.logout()
    } catch {
      // Sesi lokal tetap dibersihkan walau server menolak (mis. token sudah kedaluwarsa)
    } finally {
      loading.value = false
    }
    clearSession()
  }

  return { token, user, loading, isAuthenticated, syncToken, clearSession, login, register, logout }
})

===== src\features\cashflows\components\PeriodChart.vue =====

<script setup lang="ts">
import { computed } from 'vue'
import { formatCurrency } from '~/helpers/toolsHelper'

const props = defineProps<{
  title: string
  inflow: Record<string, number>
  outflow: Record<string, number>
}>()

const periods = computed(() =>
  Object.entries(props.inflow).map(([period, inflowValue]) => ({
    period,
    inflow: inflowValue,
    outflow: props.outflow[period] ?? 0,
  })),
)

const maxValue = computed(() =>
  Math.max(0, ...periods.value.flatMap((item) => [item.inflow, item.outflow])),
)

const barHeight = (value: number): number =>
  maxValue.value === 0 ? 0 : Math.round((value / maxValue.value) * 100)

const summary = (item: { inflow: number; outflow: number }): string =>
  `Pemasukan ${formatCurrency(item.inflow)}, pengeluaran ${formatCurrency(item.outflow)}`
</script>

<template>
  <section class="rounded-2xl border border-slate-200 bg-white p-5 shadow-sm">
    <div class="flex flex-wrap items-center justify-between gap-2">
      <h3 class="text-base font-bold text-slate-900">{{ title }}</h3>
      <div class="flex items-center gap-4 text-xs text-slate-600">
        <span class="flex items-center gap-1.5">
          <span class="h-2.5 w-2.5 rounded-full bg-income" aria-hidden="true" />
          Pemasukan
        </span>
        <span class="flex items-center gap-1.5">
          <span class="h-2.5 w-2.5 rounded-full bg-expense" aria-hidden="true" />
          Pengeluaran
        </span>
      </div>
    </div>

    <p v-if="periods.length === 0" class="py-8 text-center text-sm text-slate-500">
      Belum ada data statistik untuk ditampilkan.
    </p>

    <ul v-else class="mt-4 flex gap-2 overflow-x-auto">
      <li
        v-for="item in periods"
        :key="item.period"
        class="flex min-w-12 flex-1 flex-col items-center gap-1.5"
        :title="summary(item)"
      >
        <div class="flex h-32 w-full items-end justify-center gap-1">
          <div
            data-testid="bar-inflow"
            class="w-3 rounded-t bg-income"
            :style="{ height: `${barHeight(item.inflow)}%` }"
          />
          <div
            data-testid="bar-outflow"
            class="w-3 rounded-t bg-expense"
            :style="{ height: `${barHeight(item.outflow)}%` }"
          />
        </div>
        <span class="text-[11px] text-slate-500">{{ item.period }}</span>
        <span class="sr-only">{{ summary(item) }}</span>
      </li>
    </ul>
  </section>
</template>

===== src\features\cashflows\modals\BaseModal.vue =====

<script setup lang="ts">
import { X } from 'lucide-vue-next'
import { nextTick, ref, useId, watch } from 'vue'

const props = defineProps<{
  open: boolean
  title: string
}>()

const emit = defineEmits<{
  close: []
}>()

const titleId = `modal-title-${useId()}`
const panel = ref<HTMLElement | null>(null)

// Fokus dipindahkan ke dialog saat dibuka agar pembaca layar langsung membacakan judulnya
// dan tombol Escape langsung berfungsi
watch(
  () => props.open,
  async (open) => {
    if (!open) return
    await nextTick()
    panel.value!.focus()
  },
  { immediate: true },
)
</script>

<template>
  <div
    v-if="open"
    class="fixed inset-0 z-40 flex items-end justify-center bg-slate-900/50 p-4 sm:items-center"
    @click.self="emit('close')"
  >
    <div
      ref="panel"
      role="dialog"
      aria-modal="true"
      :aria-labelledby="titleId"
      tabindex="-1"
      class="max-h-full w-full max-w-lg overflow-y-auto rounded-2xl bg-white shadow-xl focus:outline-none"
      @keydown.esc="emit('close')"

import type { CashFlowFilters, CashFlowStats } from '~/features/cashflows/types'

/** Mengambil satu angka statistik; 0 saat kuncinya belum ada (mis. belum ada transaksi) */
export const getStat = (stats: CashFlowStats, key: string): number => stats[key] ?? 0

export const hasActiveFilters = (filters: CashFlowFilters): boolean =>
  Object.values(filters).some((value) => value.trim() !== '')

/**
 * Mengubah filter UI menjadi query API. Nilai kosong ikut dikirim sebagai string kosong
 * karena buildUrl() di apiHelper otomatis membuangnya dari URL.
 */
export const toApiQuery = (filters: CashFlowFilters): Record<string, string> => ({
  type: filters.type,
  source: filters.source,
  label: filters.label.trim(),
  start_date: filters.start_date ? `${filters.start_date} 00:00:00` : '',
  end_date: filters.end_date ? `${filters.end_date} 23:59:59` : '',
})

===== src\features\cashflows\cashflowsStore.ts =====

import { defineStore } from 'pinia'
import { computed, ref } from 'vue'
import { getStat, hasActiveFilters, toApiQuery } from '~/features/cashflows/cashflowHelper'
import { cashflowsApi } from '~/features/cashflows/cashflowsApi'
import { EMPTY_FILTERS } from '~/features/cashflows/constants'
import type {
  CashFlow,
  CashFlowFilters,
  CashFlowPayload,
  CashFlowStats,
  PeriodStats,
  StatsQuery,
} from '~/features/cashflows/types'

export const useCashFlowsStore = defineStore('cashflows', () => {
  const cashFlows = ref<CashFlow[]>([])
  const stats = ref<CashFlowStats>({})
  const filters = ref<CashFlowFilters>({ ...EMPTY_FILTERS })
  const selected = ref<CashFlow | null>(null)
  const labels = ref<string[]>([])
  const dailyStats = ref<PeriodStats | null>(null)
  const monthlyStats = ref<PeriodStats | null>(null)
  /** true saat memuat daftar atau detail transaksi */
  const loading = ref(false)
  /** true saat menambah, mengubah, atau menghapus transaksi */
  const saving = ref(false)

  const totalInflow = computed(() => getStat(stats.value, 'total_inflow'))
  const totalOutflow = computed(() => getStat(stats.value, 'total_outflow'))
  const balance = computed(() => totalInflow.value - totalOutflow.value)
  const isFiltered = computed(() => hasActiveFilters(filters.value))

  const setFilters = (next: CashFlowFilters): void => {
    filters.value = { ...next }
  }

  const resetFilters = (): void => {
    filters.value = { ...EMPTY_FILTERS }
  }

  const fetchCashFlows = async (): Promise<void> => {
    loading.value = true
    try {
      const response = await cashflowsApi.getAll(toApiQuery(filters.value))
      cashFlows.value = response.data.cash_flows
      stats.value = response.data.stats
    } finally {
      loading.value = false
    }
  }

  const fetchCashFlow = async (id: number): Promise<CashFlow> => {
    loading.value = true
    try {
      const response = await cashflowsApi.getById(id)
      selected.value = response.data.cash_flow
      return response.data.cash_flow
    } finally {
      loading.value = false
    }
  }

  const fetchLabels = async (): Promise<void> => {
    const response = await cashflowsApi.getLabels()
    labels.value = response.data.labels
  }

  const fetchDailyStats = async (query?: StatsQuery): Promise<void> => {
    const response = await cashflowsApi.getDailyStats(query)
    dailyStats.value = response.data
  }

  const fetchMonthlyStats = async (query?: StatsQuery): Promise<void> => {
    const response = await cashflowsApi.getMonthlyStats(query)
    monthlyStats.value = response.data
  }

  /** Mengembalikan id transaksi baru. Pemanggil bertugas memuat ulang daftar. */
  const createCashFlow = async (payload: CashFlowPayload): Promise<number> => {
    saving.value = true
    try {
      const response = await cashflowsApi.create(payload)
      return response.data.cash_flow_id
    } finally {
      saving.value = false
    }
  }

  const updateCashFlow = async (id: number, payload: CashFlowPayload): Promise<void> => {
    saving.value = true
    try {
      await cashflowsApi.update(id, payload)
    } finally {
      saving.value = false
    }
  }

  const deleteCashFlow = async (id: number): Promise<void> => {
    saving.value = true
    try {
      await cashflowsApi.remove(id)
    } finally {
      saving.value = false
    }
  }

  const deleteAllCashFlows = async (): Promise<void> => {
    saving.value = true
    try {
      await cashflowsApi.removeAll()
    } finally {
      saving.value = false
    }
  }

  return {
    cashFlows,
    stats,
    filters,
    selected,
    labels,
    dailyStats,
    monthlyStats,
    loading,
    saving,
    totalInflow,
    totalOutflow,
    balance,
    isFiltered,
    setFilters,
    resetFilters,
    fetchCashFlows,
    fetchCashFlow,
    fetchLabels,
    fetchDailyStats,
    fetchMonthlyStats,
    createCashFlow,
    updateCashFlow,
    deleteCashFlow,
    deleteAllCashFlows,
  }
})

/**

 * API mengembalikan URL penuh (daftar user) atau path relatif (profil saya).
 */
export const resolvePhotoUrl = (photo: string | null, baseUrl: string): string | null => {
  if (!photo) return null
  if (/^https?:\/\//i.test(photo)) return photo

  const origin = baseUrl.replace(/\/api\/v\d+\/?$/, '').replace(/\/+$/, '')
  return `${origin}/${photo.replace(/^\/+/, '')}`
}

import { defineStore } from 'pinia'

import type {
  ChangePasswordPayload,
  UpdateProfilePayload,
  UserProfile,

export const useUsersStore = defineStore('users', () => {
  const users = ref<UserProfile[]>([])
  const selectedUser = ref<UserProfile | null>(null)
  const profile = ref<UserProfile | null>(null)
  /** true saat memuat data (daftar user, detail user, profil) */
  const loading = ref(false)
  /** true saat menyimpan perubahan (profil, foto, kata sandi) */
  const saving = ref(false)

  const fetchUsers = async (): Promise<void> => {
    loading.value = true
    try {
      const response = await usersApi.getAll()
      users.value = response.data.users
    } finally {
      loading.value = false
    }
  }

  const fetchUserById = async (id: number): Promise<UserProfile> => {
    loading.value = true
    try {
      const response = await usersApi.getById(id)
      selectedUser.value = response.data.user
      return response.data.user
    } finally {
      loading.value = false
    }
  }

  const fetchProfile = async (): Promise<UserProfile> => {
    loading.value = true
    try {
      const response = await usersApi.getProfile()
      profile.value = response.data.user
      return response.data.user
    } finally {
      loading.value = false
    }
  }

  const updateProfile = async (payload: UpdateProfilePayload): Promise<void> => {
    saving.value = true
    try {
      const response = await usersApi.updateProfile(payload)
      profile.value = response.data.user
    } finally {
      saving.value = false
    }
  }

  const changePhoto = async (photo: File): Promise<void> => {
    saving.value = true
    try {
      await usersApi.changePhoto(photo)
      // Respons upload tidak membawa data user, jadi profil diambil ulang
      const response = await usersApi.getProfile()
      profile.value = response.data.user
    } finally {
      saving.value = false
    }
  }

  const changePassword = async (payload: ChangePasswordPayload): Promise<void> => {
    saving.value = true
    try {
      await usersApi.changePassword(payload)
    } finally {
      saving.value = false
    }
  }

  return {
    users,
    selectedUser,
    profile,
    loading,
    saving,
    fetchUsers,
    fetchUserById,
    fetchProfile,
    updateProfile,
    changePhoto,
    changePassword,
  }
})


===== src\helpers\apiHelper.ts =====

export const TOKEN_KEY = 'delcom_cashflow_token'
export const BASE_URL: string = import.meta.env.DELCOM_BASEURL

export type HttpMethod = 'GET' | 'POST' | 'PUT' | 'DELETE'
export type QueryValue = string | number | boolean | null | undefined

export interface ApiResponse<T = unknown> {
  status: 'success' | 'fail'
  message: string
  data: T
}

export interface RequestOptions {
  method?: HttpMethod
  body?: unknown
  query?: Record<string, QueryValue>
  auth?: boolean
}

export class ApiError extends Error {
  readonly status: number

  constructor(message: string, status: number) {
    super(message)
    this.name = 'ApiError'
    this.status = status
  }
}

export const getToken = (): string | null => localStorage.getItem(TOKEN_KEY)

export const setToken = (token: string): void => {
  localStorage.setItem(TOKEN_KEY, token)
}

export const removeToken = (): void => {
  localStorage.removeItem(TOKEN_KEY)
}

export const buildUrl = (path: string, query?: Record<string, QueryValue>): string => {
  const url = `${BASE_URL}${path}`
  if (!query) return url

  const params = new URLSearchParams()
  for (const [key, value] of Object.entries(query)) {
    if (value === undefined || value === null || value === '') continue
    params.append(key, String(value))
  }

  const queryString = params.toString()
  return queryString ? `${url}?${queryString}` : url
}

// FormData (mis. upload foto) dikirim apa adanya agar browser mengatur boundary multipart sendiri
const serializeBody = (body: unknown): BodyInit | undefined => {
  if (body === undefined) return undefined
  return body instanceof FormData ? body : JSON.stringify(body)
}

export async function request<T = unknown>(
  path: string,
  options: RequestOptions = {},
): Promise<ApiResponse<T>> {
  const { method = 'GET', body, query, auth = true } = options

  const headers: Record<string, string> = { Accept: 'application/json' }
  if (body !== undefined && !(body instanceof FormData)) {
    headers['Content-Type'] = 'application/json'
  }

  const token = getToken()
  if (auth && token) headers.Authorization = `Bearer ${token}`

  let response: Response
  try {
    response = await fetch(buildUrl(path, query), {
      method,
      headers,
      body: serializeBody(body),
    })
  } catch {
    throw new ApiError('Tidak dapat terhubung ke server. Periksa koneksi internet Anda.', 0)
  }

  let payload: ApiResponse<T> | null
  try {
    payload = (await response.json()) as ApiResponse<T>
  } catch {
    payload = null
  }

  if (response.status === 401 && auth) removeToken()

  if (!response.ok || payload === null || payload.status !== 'success') {
    throw new ApiError(payload?.message ?? `Permintaan gagal (${response.status})`, response.status)
  }

  return payload
}

export const api = {
  get: <T = unknown>(path: string, query?: Record<string, QueryValue>) =>
    request<T>(path, { method: 'GET', query }),
  post: <T = unknown>(path: string, body?: unknown, options?: Pick<RequestOptions, 'auth'>) =>
    request<T>(path, { ...options, method: 'POST', body }),
  put: <T = unknown>(path: string, body?: unknown) =>
    request<T>(path, { method: 'PUT', body }),
  delete: <T = unknown>(path: string) => request<T>(path, { method: 'DELETE' }),
}

===== src\hooks\useInput.ts =====

import { ref } from 'vue'
import type { Ref } from 'vue'

export interface UseInputReturn<T extends string | number> {
  value: Ref<T>
  onInput: (event: Event) => void
  reset: () => void
}

type InputElement = HTMLInputElement | HTMLTextAreaElement | HTMLSelectElement

export function useInput<T extends string | number = string>(initialValue: T): UseInputReturn<T> {
  const value = ref(initialValue) as Ref<T>

  const onInput = (event: Event): void => {
    const raw = (event.target as InputElement).value
    value.value = (typeof initialValue === 'number' ? Number(raw) : raw) as T
  }

  const reset = (): void => {
    value.value = initialValue
  }

  return { value, onInput, reset }
}

  const { initialState, stubActions = true, props, slots, router, global = {} } = options

    initialState,
    stubActions,

  })

    props,
    slots,
    global: {
      ...global,
      plugins: [pinia, ...(router ? [router] : []), ...(global.plugins ?? [])],
      stubs: {
        ...(router ? {} : { RouterLink: RouterLinkStub, NuxtLink: RouterLinkStub }),
        ...global.stubs,
      },
    },
  })
